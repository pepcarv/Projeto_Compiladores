2. GRAMATICA LIVRE DE CONTEXTO (GLC)
-----------------------------------------------------------------------------
Construir a GLC correspondente as expressoes regulares acima, cobrindo TODAS
as estruturas exemplificadas no Anexo I. Sugestao de nao-terminais a definir
(preencher as producoes de cada um):

<programa> ::= "algoritmo" <cadeia>
                { <declaracao_procedimento> | <declaracao_funcao> }
                [ "var" <declaracoes> ]
                "inicio" <lista_comandos>
                "fimalgoritmo"

<declaracoes> ::= <declaracao> { <declaracao> }
<declaracao> ::= <lista_identificadores> ":" <tipo>
<lista_identificadores> ::= <identificador>
                            { "," <identificador> }

<tipo> ::= <tipo_tipo>
         | "vetor" "[" <inteiro> ".." <inteiro> "]" "de" <tipo_tipo>
<tipo_tipo> ::= "caractere"
                | "inteiro"
                | "real"
                | "logico"

<bloco_comandos> ::= { <comando> }

<comando> ::= <atribuicao>
            | <leitura>
            | <escrita>
            | <condicional>
            | <repeticao_para>
            | <repeticao_enquanto>
            | <retorno>
            | <chamada>

<atribuicao> ::= <id> [ "[" <expressao> "]" ] "<-" <expressao>

<leitura> ::= "leia" "(" <id> [ "[" <expressao> "]" ] ")"

<escrita> ::= ( "escreva" | "escreval" ) "(" <lista_expressoes> ")"
<lista_expressoes> ::= <expressao> { "," <expressao> }

<condicional> ::= "se" "(" <expressao_relacional> ")" "entao"
                  <bloco_comandos>
                  [ "senao" <bloco_comandos> ]
                  "fimse"

<repeticao_para> ::= "para" <id> "de" <expressao>
                      "ate" <expressao>
                      [ "passo" <expressao> ]
                      "faca" <bloco_comandos> "fimpara"

<repeticao_enquanto> ::= "enquanto" "(" <expressao_relacional> ")"
                          "faca" <bloco_comandos> "fimenquanto"

<declaracao_procedimento> ::= "procedimento" <id>
                              "inicio" <bloco_comandos>
                              "fimprocedimento"
                            | "procedimento" <id>
                              "(" <lista_parametros> ")"
                              "inicio" <bloco_comandos>
                              "fimprocedimento"
<lista_parametros> ::= <parametro> { "," <parametro> }
<parametro> ::= <id> ":" <tipo_tipo>

<declaracao_funcao> ::= "funcao" <id>
                        "(" <lista_parametros> ")"
                        ":" <tipo_tipo>
                        "inicio" <bloco_comandos>
                        "fimfuncao"

<retorno> ::= "retorne" <expressao>

<chamada> ::= <id>
            | <id> "(" <lista_argumentos> ")"
<lista_argumentos> ::= <expressao> { "," <expressao> }


<expressao> ::= <expressao_relacional> { <operador_logico> <expressao_relacional> }

<expressao_relacional> ::= <expressao_aritmetica>
                          [ <operador_relacional> <expressao_aritmetica> ]
<operador_relacional> ::= "=" | "<>" | "<" | "<=" | ">" | ">="
<operador_logico> ::= "E" | "OU"
<expressao_aritmetica> ::= <termo> { ( "+" | "-" ) <termo> }
<termo> ::= <fator> { ( "*" | "/" | "\\" | "MOD" ) <fator> }



<fator> ::= <numero_inteiro>
          | <numero_real>
          | <string>
          | "verdadeiro"
          | "falso"
          | <id> [ "[" <expressao> "]" ]
          | <chamada_funcao>
          | "(" <expressao> ")"
<chamada_funcao> ::= <id> "(" <lista_argumentos> ")"
<id> ::= token identificador
<numero_inteiro> ::= token numero inteiro
<numero_real> ::= token numero real
<string> ::= token cadeia de caracteres
