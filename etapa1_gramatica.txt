2. GRAMATICA LIVRE DE CONTEXTO (GLC)
-----------------------------------------------------------------------------


<programa> ::= "algoritmo" <string> { <declaracao_rotina> }
                [ "var" <declaracoes> ]
                "inicio" <bloco_comandos> "fimalgoritmo"

TODO: <declaracoes> ::=

  lista de: <lista_identificadores> ":" <tipo>
  (ver secao VARIAVEIS: "nome, sobrenome: caractere")





<tipo> ::= "caractere"
          | "inteiro"
          | "real"
          | "logico"
          | "vetor" "[" <inteiro> ".." <inteiro> "]" "de" <tipo>

<bloco_comandos> ::= { <comando> }

<comando> ::= <atribuicao>
            | <leitura>
            | <escrita>
            | <condicional>
            | <repeticao_para>
            | <repeticao_enquanto>
            | <retorno>
            | <chamada>

<atribuicao> ::= <id> [ "[" <expressao> "]" ] "<-" <expressao>


<leitura> ::= "leia" "(" <id> [ "[" <expressao> "]" ] ")"

<escrita> ::= ( "escreva" | "escreval" ) "(" <lista_expressoes> ")"




<condicional> ::= "se" "(" <expressao> ")" "entao"
                  <bloco_comandos>
                  [ "senao" <bloco_comandos> ]
                  "fimse"


<repeticao_para> ::= "para" <id> "de" <expressao> "ate" <expressao>
                      [ "passo" <expressao> ]
                      "faca" <bloco_comandos> "fimpara"

<repeticao_enquanto> ::= "enquanto" "(" <expressao> ")" "faca"
                          <bloco_comandos> "fimenquanto"

<declaracao_procedimento> ::= "procedimento" <id>
                              [ "(" <lista_parametros> ")" ]
                              [ "var" <declaracoes> ]
                              "inicio" <bloco_comandos> "fimprocedimento"

<declaracao_funcao> ::= "funcao" <id> "(" [ <lista_parametros> ] ")"
                        ":" <tipo>
                        [ "var" <declaracoes> ]
                        "inicio" <bloco_comandos> "fimfuncao"

<chamada> ::= <id> [ "(" [ <lista_argumentos> ] ")" ]

<expressao> ::= <termo_ou>

<expressao_relacional> ::= <expressao_aritmetica>
                           [ <operador_relacional> <expressao_aritmetica> ]

<fator> ::= <inteiro>
          | <real>
          | <string>
          | "verdadeiro"
          | "falso"
          | <id> [ "[" <expressao> "]" ] [ "(" [ <lista_argumentos> ] ")" ]
          | "(" <expressao> ")"
